4.2 JWT Utility (TOKEN ENGINE)
@Component
public class JwtUtil {

    private final String SECRET = "very_secure_key_123456789";

    public String generateToken(String username, String role) {
        return Jwts.builder()
          .setSubject(username)
          .claim("role", role)
          .setIssuedAt(new Date())
          .setExpiration(new Date(System.currentTimeMillis()+86400000))
          .signWith(Keys.hmacShaKeyFor(SECRET.getBytes()))
          .compact();
    }
}

4.3 Security Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {

        http.csrf().disable()
          .authorizeHttpRequests(auth -> auth
            .requestMatchers("/auth/**").permitAll()
            .requestMatchers("/admin/**").hasRole("ADMIN")
            .requestMatchers("/tracker/**").hasRole("TRACKER")
            .anyRequest().authenticated()
          );

        return http.build();
    }
}

4.4 Login API
@RestController
@RequestMapping("/auth")
public class AuthController {

    @Autowired JwtUtil jwtUtil;

    @PostMapping("/login")
    public String login(@RequestBody LoginRequest req) {
        return jwtUtil.generateToken(req.getUsername(), req.getRole());
    }
}

📦 STEP 5: DTO-BASED LOCATION HANDLING (SECURE)
Why DTO?

❌ Never expose entity directly
✅ Prevent DB manipulation

Location Entity
@Entity
@Table(name="locations")
public class Location {

    @Id @GeneratedValue
    private Long id;

    private Long userId;
    private double latitude;
    private double longitude;
    private LocalDateTime timestamp;
}

Location DTO
public class LocationDTO {
    private Long userId;
    private double latitude;
    private double longitude;
}

Mapper
public class LocationMapper {

    public static Location toEntity(LocationDTO dto) {
        Location loc = new Location();
        loc.setUserId(dto.getUserId());
        loc.setLatitude(dto.getLatitude());
        loc.setLongitude(dto.getLongitude());
        loc.setTimestamp(LocalDateTime.now());
        return loc;
    }
}

⚡ STEP 6: REAL-TIME WEBSOCKET
WebSocket Config
@Configuration
@EnableWebSocketMessageBroker
public class WebSocketConfig implements WebSocketMessageBrokerConfigurer {

    public void configureMessageBroker(MessageBrokerRegistry config) {
        config.enableSimpleBroker("/topic");
        config.setApplicationDestinationPrefixes("/app");
    }

    public void registerStompEndpoints(StompEndpointRegistry registry) {
        registry.addEndpoint("/track")
          .setAllowedOriginPatterns("*")
          .withSockJS();
    }
}

WebSocket Controller (TRACKER ONLY)
@Controller
public class TrackingController {

    @Autowired LocationService service;
    @Autowired SimpMessagingTemplate messaging;

    @MessageMapping("/location")
    @PreAuthorize("hasRole('TRACKER')")
    public void track(LocationDTO dto) {

        Location saved = service.save(dto);
        messaging.convertAndSend("/topic/location", saved);
    }
}

📜 STEP 7: REST API FOR HISTORY (ADMIN)
@RestController
@RequestMapping("/admin")
public class HistoryController {

    @Autowired LocationRepository repo;

    @GetMapping("/history/{userId}")
    public List<Location> getHistory(@PathVariable Long userId) {
        return repo.findByUserIdOrderByTimestampDesc(userId);
    }
}
